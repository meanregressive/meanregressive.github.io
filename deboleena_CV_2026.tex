% Heading font: ~10% larger than the 11pt body, reused by \cvheading

\newcommand{\cvheadfont}{\fontsize{12}{14.5}\selectfont}

% One job entry: {Organisation}{Location}{Role / titles}{Dates}
% Typeset as a standalone paragraph (NOT a list item), so it sits flush at
% the text margin, aligned with the section rule. \noindent + @{} on the
% tabular* outer edges keep both lines flush left; \extracolsep{\fill}
% pushes the location/dates to the right margin.
% \\[3pt] = extra gap between the org line and the role/dates line.
\newcommand{\cvheading}[4]{%
  \vspace{2mm}\par\noindent
  \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
    {\cvheadfont\textbf{#1}} & \textit{\footnotesize #2} \\[3pt]
    \textit{\footnotesize #3} & \footnotesize #4 \\
  \end{tabular*}\par\vspace{-0.4mm}
}

% Single-line heading in the SAME style as \cvheading's org name (same font,
% weight, flush-left margin, and leading space). Used for the publication
% subsection headings so they match the experience headings exactly.
\newcommand{\cvsubheading}[1]{\vspace{2mm}\par\noindent{\cvheadfont\textbf{#1}}\par}

\newcommand{\substart}{\begin{itemize}[leftmargin=*,labelsep=1mm,itemsep=0.2mm,topsep=0.3mm,nosep,label=\textasteriskcentered]}

% Bulleted detail under a job: {Bold lead}{text}
\newcommand{\cvitem}[2]{\item \textbf{#1} #2}

% References line for a role: {names / what they speak to}. No bullet; pulled
% back out of the bullet indent so it sits flush at the text margin, aligned
% with \cvheading. The \parbox keeps any wrapped lines flush left too.

  \item[]\vspace{0.4mm}%
  \hspace*{-\leftmargin}\parbox[t]{\dimexpr\linewidth+\leftmargin\relax}{%

}
